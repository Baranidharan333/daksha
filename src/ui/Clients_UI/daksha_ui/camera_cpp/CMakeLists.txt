cmake_minimum_required(VERSION 3.16)
project(viveka_camera_native VERSION 1.0.0 LANGUAGES CXX)
find_package(PkgConfig REQUIRED)
find_package(Threads REQUIRED)
find_package(Boost REQUIRED CONFIG)
find_package(nlohmann_json 3.2.0 REQUIRED)
pkg_check_modules(GSTREAMER REQUIRED IMPORTED_TARGET
  gstreamer-1.0 gstreamer-webrtc-1.0 gstreamer-sdp-1.0 gstreamer-video-1.0)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS static/index.html)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/static/index.html" VIVEKA_PAGE)
configure_file(src/camera_page.hpp.in generated/camera_page.hpp @ONLY)
add_executable(viveka_camera_ui src/main.cpp)
target_compile_features(viveka_camera_ui PRIVATE cxx_std_17)
target_include_directories(viveka_camera_ui PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated" ${Boost_INCLUDE_DIRS})
target_link_libraries(viveka_camera_ui PRIVATE PkgConfig::GSTREAMER Threads::Threads nlohmann_json::nlohmann_json)
target_compile_definitions(viveka_camera_ui PRIVATE BOOST_ERROR_CODE_HEADER_ONLY BOOST_SYSTEM_NO_DEPRECATED GST_USE_UNSTABLE_API)
target_compile_options(viveka_camera_ui PRIVATE -Wall -Wextra -Wpedantic)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  install(TARGETS viveka_camera_ui RUNTIME DESTINATION bin)
else()
  install(TARGETS viveka_camera_ui RUNTIME DESTINATION lib/daksha_ui)
endif()
